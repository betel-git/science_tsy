\begin{frame}{План доклада}
  \begin{enumerate}
    \item Онлайн-прогнозирование и сравнение с экспертами.
    \item Экспоненциальные веса и адаптивный параметр обучения.
    \item Спящие эксперты и алгоритм sAdaHedge.
    \item Оценка регрета через дисперсию потерь.
    \item Эксперимент с суточным электропотреблением.
  \end{enumerate}
\end{frame}
\begin{frame}{Временной ряд, эксперты и агрегатор}
  \begin{block}{Определение 1. Временной ряд}
    Временной ряд --- последовательность исходов
    $y=(y_1,y_2,\ldots)$, где $y_t\in\mathcal Y$ и $t$ обозначает время.
  \end{block}
  \begin{block}{Определение 2. Экспертная стратегия}
    Эксперт $i$ --- правило, выдающее по доступной истории прогноз
    $\gamma_t^i\in\mathcal D$ до наблюдения $y_t$.
    Здесь $\mathcal D$ --- множество допустимых прогнозов.
  \end{block}
  \begin{block}{Определение 3. Агрегатор}
    Агрегатор --- алгоритм, который по прогнозам экспертов и прошлым
    наблюдениям строит собственный прогноз $\gamma_t\in\mathcal D$.
  \end{block}
  \par\vfill{\scriptsize Экспертами могут быть регрессионные модели, сезонные правила и оценки специалистов.\par}
\end{frame}
\begin{frame}{Протокол онлайн-прогнозирования}
  Эксперты: $I=\{1,\ldots,N\}$. Исходы: $y_t\in\mathcal Y$.
  Прогнозы: $\gamma_t^i,\gamma_t\in\mathcal D$.
  \medskip
  \begin{enumerate}
    \item По истории до шага $t$ эксперты выдают $\gamma_t^1,\ldots,\gamma_t^N$.
    \item Агрегатор выбирает собственный прогноз $\gamma_t$.
    \item Среда раскрывает исход $y_t$.
    \item Вычисляются потери. Агрегатор обновляет своё состояние.
  \end{enumerate}
  \par\medskip\textbf{Прогноз и веса выбираются до раскрытия текущего исхода.}
  \par\vfill{\scriptsize Вероятностную модель ряда не предполагаем. Гарантия должна работать для любой последовательности исходов.\par}
\end{frame}
\begin{frame}{Функции потерь}
  \begin{block}{Определение 4. Функция потерь}
    Функция $\ell:\mathcal D\times\mathcal Y\to[0,+\infty)$
    сопоставляет прогнозу $\gamma$ и исходу $y$ величину ошибки $\ell(\gamma,y)$.
  \end{block}
  \begin{align*}
    \text{Квадратичная:}\qquad &\ell(\gamma,y)=(\gamma-y)^2,\\[2mm]
    \text{Абсолютная:}\qquad &\ell(\gamma,y)=|\gamma-y|,\\[2mm]
    \text{Несимметричная:}\qquad
      &\ell(\gamma,y)=\lambda_1(\gamma-y)_++\lambda_2(y-\gamma)_+.
  \end{align*}
  Здесь $(x)_+=\max\{x,0\}$, $\lambda_1,\lambda_2>0$.
  При $\lambda_2>\lambda_1$ недооценка штрафуется сильнее переоценки.
  \par\vfill{\scriptsize Все три примера выпуклы по прогнозу $\gamma$. Это свойство понадобится при усреднении прогнозов.\par}
\end{frame}
\begin{frame}{Кумулятивные потери и регрет}
  \begin{block}{Определение 5. Кумулятивные потери}
    Суммарные потери на первых $T$ шагах:
    \[
      L_T:=\sum_{t=1}^T\ell(\gamma_t,y_t),\qquad
      L_T^i:=\sum_{t=1}^T\ell(\gamma_t^i,y_t).
    \]
  \end{block}
  \begin{block}{Определение 6. Регрет}
    Регрет относительно лучшего фиксированного эксперта:
    \[R_T:=L_T-\min_{i\in I}L_T^i.\]
  \end{block}
  \begin{itemize}
    \item Лучший эксперт выбирается после всех $T$ шагов; его индекс фиксирован.
    \item Цель: малый добавочный проигрыш, в частности $R_T/T\to0$.
  \end{itemize}
  \par\vfill{\scriptsize Малый регрет сам по себе не означает малой абсолютной ошибки: важен уровень качества экспертов.\par}
\end{frame}

\section{Экспоненциальное смешивание}
\begin{frame}{Задача смешивания потерь}
  \begin{block}{Определение 7. Распределение на множестве экспертов}
    Вектор весов $p_t=(p_t^1,\ldots,p_t^N)$, для которого
    $p_t^i\ge0$ и $\sum_{i=1}^N p_t^i=1$.
  \end{block}
  \begin{block}{Определение 8. Смесь потерь}
    Средняя потеря экспертов с весами $p_t$ и её накопленное значение:
    \[
      \ell_t:=\sum_{i=1}^N p_t^i\ell_t^i,\qquad
      L_T:=\sum_{t=1}^T\ell_t.
    \]
  \end{block}
  Веса выбираются до раскрытия текущих потерь $\ell_t^i$.
  \par\smallskip
  Здесь достаточно последовательности чисел $\ell_t^i$:
  конкретная функция потерь может не задаваться.
  \par\vfill{\scriptsize В классической постановке можно считать $\ell_t^i\in[0,1]$. Позже оценка будет зависеть от фактического размаха потерь.\par}
\end{frame}

\begin{frame}{Hedge: экспоненциальные веса}
  Фиксируем $\eta>0$ и начальные веса $w_1^i=1/N$.
  \begin{enumerate}
    \item Нормируем веса: $\displaystyle p_t^i=\frac{w_t^i}{\sum_j w_t^j}$.
    \item Получаем потери и считаем $\displaystyle\ell_t=\sum_i p_t^i\ell_t^i$.
    \item Обновляем веса: $\displaystyle w_{t+1}^i=w_t^i e^{-\eta\ell_t^i}$.
  \end{enumerate}
  \medskip
  После раскрытия формулы обновления:
  \[
    w_t^i=\frac1N e^{-\eta L_{t-1}^i},\qquad
    p_t^i=\frac{e^{-\eta L_{t-1}^i}}{\sum_j e^{-\eta L_{t-1}^j}}.
  \]
  \par\medskip\textbf{Меньшие накопленные потери дают больший вес.}
  %\par\vfill{\scriptsize Экспоненциальное взвешивание: Freund--Schapire~\cite{FreundSchapire}.\par}
\end{frame}

\begin{frame}{Смешанная потеря и смешанная разность}
  \begin{block}{Определение 9. Смешанная потеря}
    При параметре $\eta>0$ смешанная потеря и её сумма равны
    \[
      m_t:=-\frac1\eta\ln\!\left(\sum_i p_t^i e^{-\eta\ell_t^i}\right),
      \qquad M_T:=\sum_{t=1}^T m_t.
    \]
  \end{block}
  \begin{block}{Определение 10. Смешанная разность}
    \[\delta_t:=\ell_t-m_t,\qquad \Delta_T:=\sum_{t=1}^T\delta_t.\]
  \end{block}
  \textbf{Свойства:}
  \[\min_i\ell_t^i\le m_t\le\ell_t\le\max_i\ell_t^i,\qquad L_T=M_T+\Delta_T.\]
  \par\vfill{\scriptsize Неравенство Йенсена даёт $m_t\le\ell_t$, поэтому $\delta_t\ge0$.\par}
\end{frame}
\begin{frame}{Потенциал и сравнение с экспертом}
  Положим $W_t=\sum_i w_t^i$. При постоянном $\eta$:
  \[
    \sum_i p_t^i e^{-\eta\ell_t^i}=\frac{W_{t+1}}{W_t},\qquad
    M_T=-\frac1\eta\sum_{t=1}^T\ln\frac{W_{t+1}}{W_t}=-\frac1\eta\ln W_{T+1}.
  \]
  Так как $W_1=1$ и $W_{T+1}\ge\frac1N e^{-\eta L_T^i}$,
  \[M_T\le L_T^i+\frac{\ln N}{\eta}\qquad\text{для любого }i.\]
  \begin{block}{Разложение регрета}
    \[L_T-L_T^i=(M_T-L_T^i)+\Delta_T\le\frac{\ln N}{\eta}+\Delta_T.\]
  \end{block}
\end{frame}
\begin{frame}{Переменный параметр обучения}
  При каждом новом $\eta_t$ пересчитываем веса по всей истории:
  \[w_t^i=\frac1N e^{-\eta_t L_{t-1}^i},\qquad p_t^i=\frac{w_t^i}{\sum_j w_t^j}.\]
  Обозначим через $M_t^\eta$ потенциал с постоянным параметром $\eta$.
  Тогда
  \[m_t=M_t^{\eta_t}-M_{t-1}^{\eta_t}.\]
  \begin{block}{Если $\eta_1\ge\eta_2\ge\cdots\ge\eta_T>0$}
    \[M_T=\sum_{t=1}^T m_t\le M_T^{\eta_T}\le L_T^i+\frac{\ln N}{\eta_T}.\]
  \end{block}
  %\par\vfill{\scriptsize Используется невозрастание $M_t^\eta$ по $\eta$. Доказательства: резерв A1--A2. Адаптивные веса: \cite{AdaHedge}.\par}
\end{frame}

\section{Специалисты и sAdaHedge}
\begin{frame}{Эксперты-специалисты}
  \begin{block}{Определение 11. Эксперт-специалист}
    Специалист --- эксперт, участвующий в прогнозировании лишь на части шагов.
    На шаге $t$ множество $E_t\subseteq I$ содержит активных экспертов;
    эксперты из $I\setminus E_t$ называются спящими.
  \end{block}
  \begin{itemize}
    \item Специализация может зависеть от дня недели, сезона или режима ряда.
    \item Эксперт может отсутствовать из-за недоступных признаков.
    \item Его качество следует оценивать на шагах, когда он участвовал.
  \end{itemize}
  \par\medskip\textbf{Условия протокола.}
    $N\ge2$, $E_t\ne\varnothing$. Множество $E_t$ определяется
    до раскрытия текущего исхода или текущих потерь.
  %\par\vfill{\scriptsize Специализированное агрегирование: \cite{VyuginSham,Devaine}. В рассматриваемой теореме множество $I$ фиксировано.\par}
\end{frame}

\begin{frame}{Нормировка на активных экспертах}
  Глобальные веса хранят информацию обо всех экспертах:
  \[
    w_t^i=\frac1N e^{-\eta_t L_{t-1}^i},\qquad
    p_t^i=\frac{w_t^i}{\sum_j w_t^j}.
  \]
  \begin{block}{Определение 12. Нормированные веса активных экспертов}
    Распределение весов на непустом множестве $E_t$:
    \[
       p_t^{i,E_t}:=\frac{p_t^i}{\sum_{j\in E_t}p_t^j},\qquad i\in E_t.
    \]
  \end{block}
  Потеря смеси теперь учитывает только активных экспертов:
  \[\ell_t=\sum_{i\in E_t} p_t^{i,E_t}\ell_t^i.\]
  \par\vfill{\scriptsize При конечном $\eta_t$ веса положительны. Случай $\eta_t=\infty$ рассмотрим отдельно.\par}
\end{frame}

\begin{frame}{Виртуальные потери спящих экспертов}
  \begin{block}{Определение 13. Виртуальные потери в sAdaHedge}
    На шагах сна эксперту приписывается смешанная потеря агрегатора:
    $\ell_t^i:=m_t$ при $i\notin E_t$.
  \end{block}
  \begin{block}{Определение 14. Расширенные кумулятивные потери}
    К фактическим потерям на активных шагах добавляются виртуальные потери:
    \[
      \mathcal L_T^i:=\sum_{\substack{1\le t\le T\\i\in E_t}}\ell_t^i,
      \qquad
      L_T^i:=\mathcal L_T^i+\sum_{\substack{1\le t\le T\\i\notin E_t}}m_t.
    \]
  \end{block}
  Каждый эксперт получает последовательность потерь на всех шагах.
  К ней можно применить оценки экспоненциальных весов.
  \par\vfill{\scriptsize Виртуальная потеря равна $m_t$; скрытая фактическая потеря спящего эксперта в обновлении не используется.\par}
\end{frame}

\begin{frame}{Как вычислить смешанную потерю}
  Пусть $q_t=\sum_{i\in E_t}p_t^i>0$. Подставим виртуальные потери:
  \[
    e^{-\eta_t m_t}
    =\sum_{i\in E_t}p_t^i e^{-\eta_t\ell_t^i}
      +(1-q_t)e^{-\eta_t m_t}.
  \]
  Перенос последнего слагаемого даёт
  \[
    q_t e^{-\eta_t m_t}=\sum_{i\in E_t}p_t^i e^{-\eta_t\ell_t^i}.
  \]
  \begin{block}{Формула через доступные потери}
    \[
      m_t=-\frac1{\eta_t}\ln\!\left(\sum_{i\in E_t} p_t^{i,E_t} e^{-\eta_t\ell_t^i}\right).
    \]
  \end{block}
  \par\vfill{\scriptsize Правая часть зависит только от доступных потерь и известных весов: цикличности нет.\par}
\end{frame}

\begin{frame}{Адаптивный параметр sAdaHedge}
  Как и раньше, определим
  \[
    \delta_t=\ell_t-m_t\ge0,\qquad
    \Delta_t=\Delta_{t-1}+\delta_t,\qquad\Delta_0=0.
  \]
  Параметр обучения выбирается до текущих потерь:
  \[
    \eta_t=\begin{cases}
      \dfrac{\ln N}{\Delta_{t-1}},&\Delta_{t-1}>0,\\[2mm]
      \infty,&\Delta_{t-1}=0.
    \end{cases}
  \]
  \begin{itemize}
    \item $\Delta_t$ не убывает, поэтому $\eta_t$ не возрастает.
    \item Горизонт $T$ и будущий размах потерь заранее не нужны.
    \item При большем накопленном разрыве веса менее сосредоточены.
  \end{itemize}
  %\par\vfill{\scriptsize sAdaHedge: Вьюгин--Шамсутдинов~\cite{VyuginSham}; адаптация через смешанную разность: \cite{AdaHedge}.\par}
\end{frame}

\begin{frame}{Начальный случай: смешанная разность равна нулю}
  Инициализация: $L_0^i=0$, $w_1^i=1/N$, $\Delta_0=0$.
  \medskip
  Пока $\Delta_{t-1}=0$, расширенные потери всех экспертов совпадают.
  Используем предельное соглашение:
  \[
    \eta_t=\infty,\qquad \frac1{\eta_t}=0,\qquad
    p_t^i=\frac1N,\qquad p_t^{i,E_t}=\frac1{|E_t|}.
  \]
  \[
    m_t=\min_{i\in E_t}\ell_t^i,\qquad
    \delta_t=\frac1{|E_t|}\sum_{i\in E_t}\ell_t^i-\min_{i\in E_t}\ell_t^i.
  \]
  \par\medskip\textbf{При первом различии активных потерь получаем $\delta_t>0$.}
  На следующем шаге $\eta_{t+1}$ уже конечен.
  \par\vfill{\scriptsize Если все активные потери равны, виртуальные потери совпадают с ними; равенство расширенных потерь сохраняется.\par}
\end{frame}

\begin{frame}{Один шаг sAdaHedge}
  \textbf{До раскрытия потерь}
  \begin{enumerate}
    \item Получить $E_t$. Вычислить $\eta_t$ по $\Delta_{t-1}$.
    \item По $L_{t-1}^i$ пересчитать веса и нормировать их на $E_t$.
    \item При прогнозировании выдать $\displaystyle\gamma_t=\sum_{i\in E_t} p_t^{i,E_t}\gamma_t^i$.
  \end{enumerate}
  \medskip
  \textbf{После раскрытия потерь}
  \begin{enumerate}
    \setcounter{enumi}{3}
    \item Вычислить $\ell_t$, $m_t$, $\delta_t=\ell_t-m_t$.
    \item Спящим экспертам присвоить $\ell_t^i=m_t$.
    \item Обновить $L_t^i=L_{t-1}^i+\ell_t^i$ и $\Delta_t=\Delta_{t-1}+\delta_t$.
  \end{enumerate}
  \par\vfill{\scriptsize На следующем шаге веса вычисляются заново с новым $\eta_{t+1}$. При переменном параметре простой множитель $e^{-\eta_t\ell_t^i}$ не заменяет этот пересчёт.\par}
\end{frame}

\section{Гарантия качества}
\begin{frame}{Регрет относительно специалиста}
  \begin{block}{Определение 15. Регрет специалиста}
    Разность потерь смеси и эксперта на шагах активности этого эксперта:
    \[
      \widetilde R_T^i:=
         \sum_{\substack{1\le t\le T\\i\in E_t}}(\ell_t-\ell_t^i).
    \]
  \end{block}
  \begin{block}{Определение 16. Расширенный регрет}
    Регрет по расширенным потерям на всём горизонте:
    $R_T^i:=L_T-L_T^i$.
  \end{block}
  Во время сна $\ell_t^i=m_t$, поэтому
  \[
    R_T^i=\widetilde R_T^i+
       \sum_{\substack{1\le t\le T\\i\notin E_t}}\delta_t
       \ge\widetilde R_T^i.
  \]
  \par\vfill{\scriptsize Для разных специалистов наборы активных дней различаются; $\delta_t\ge0$ даёт сравнение регретов.\par}
\end{frame}

\begin{frame}{Дисперсия и размах потерь}
  \begin{block}{Определение 17. Размах потерь}
    Разность максимальной и минимальной потерь активных экспертов:
    \[
      s_t:=\max_{i\in E_t}\ell_t^i-\min_{i\in E_t}\ell_t^i,
      \qquad S_T:=\max_{1\le t\le T}s_t.
    \]
  \end{block}
  \begin{block}{Определение 18. Взвешенная дисперсия потерь}
    Дисперсия относительно распределения активных весов и её сумма:
    \[
      v_t:=\sum_{i\in E_t} p_t^{i,E_t}(\ell_t^i-\ell_t)^2,
      \qquad V_T:=\sum_{t=1}^T v_t.
    \]
  \end{block}
  \begin{itemize}
    \item Из границ смешанной потери следует $0\le\delta_t\le s_t$.
    \item При совпадении всех активных потерь $s_t=v_t=\delta_t=0$.
  \end{itemize}
  \par\vfill{\scriptsize Дисперсия определяется по весам экспертов; вероятностная модель ряда не требуется.\par}
\end{frame}

\begin{frame}{Оценка регрета sAdaHedge}
  \begin{block}{Теорема Вьюгина--Шамсутдинова}
    При $N\ge2$, непустых $E_t$ и описанном протоколе для любого
    эксперта $i$ и любого $T\ge1$ выполнено
    \[
      \widetilde R_T^i\le
      2\sqrt{V_T\ln N}+S_T\left(\frac43\ln N+2\right).
    \]
  \end{block}
  \medskip
  \begin{itemize}
    \item Одна граница одновременно для всех специалистов.
    \item Сложность последовательности входит через $V_T$ и $S_T$.
    \item При $s_t\le1$ имеем $V_T\le T/4$ и порядок
          $O(\sqrt{T\ln N}+\ln N)$.
  \end{itemize}
  %\par\vfill{\scriptsize Вьюгин--Шамсутдинов (2016)~\cite{VyuginSham}. Следующие слайды раскрывают ключевые переходы доказательства.\par}
\end{frame}

\begin{frame}{Шаг 1. Сведение регрета к смешанной разности}
  Применим оценку с невозрастающим $\eta_t$ к расширенным потерям:
  \[
    M_T-L_T^i\le\frac{\ln N}{\eta_T}
      =\Delta_{T-1}\le\Delta_T.
  \]
  Используя $L_T=M_T+\Delta_T$, получаем
  \[
    R_T^i=L_T-L_T^i=(M_T-L_T^i)+\Delta_T\le2\Delta_T.
  \]
  \begin{block}{Промежуточная гарантия}
    \[\widetilde R_T^i\le R_T^i\le2\Delta_T.\]
  \end{block}
  Осталось оценить $\Delta_T$ через наблюдаемую дисперсию и размах.
  \par\vfill{\scriptsize При $\Delta_{T-1}=0$ используется соглашение $1/\eta_T=0$ и тот же предельный смысл потенциала.\par}
\end{frame}

\begin{frame}{Шаг 2. Одношаговая оценка через дисперсию}
  Для $s_t>0$ и конечного $\eta_t$ неравенство Бернштейна даёт
  \[
    \delta_t\le
    \frac{e^{s_t\eta_t}-1-s_t\eta_t}{\eta_t s_t^2}\,v_t.
  \]
  Для суммирования удобнее следующее следствие:
  \begin{block}{Вспомогательное неравенство}
    \[\frac{\delta_t}{\eta_t}\le\frac12v_t+\frac13s_t\delta_t.\]
  \end{block}
  \begin{itemize}
    \item Оно верно при любом конечном $\eta_t>0$.
    \item При $s_t=0$ обе части равны нулю.
    \item При $\eta_t=\infty$ левая часть равна нулю.
  \end{itemize}
  %\par\vfill{\scriptsize Полное применение Бернштейна и переход для малых и больших $s_t\eta_t$: резерв A4--A5.\par}
\end{frame}

\begin{frame}{Шаг 3. Накопление смешанных разностей}
  Телескопируем квадраты и используем $\Delta_{t-1}=\ln N/\eta_t$:
  \begin{align*}
    \Delta_T^2
      &=\sum_{t=1}^T\left(2\delta_t\Delta_{t-1}+\delta_t^2\right)\\
      &\le2\ln N\sum_{t=1}^T\frac{\delta_t}{\eta_t}+S_T\Delta_T\\
      &\le2\ln N\sum_{t=1}^T\left(\frac12v_t+\frac13s_t\delta_t\right)
           +S_T\Delta_T.
  \end{align*}
  Здесь $\delta_t^2\le s_t\delta_t$ и $s_t\le S_T$.
  \[\alert{\Delta_T^2\le V_T\ln N+\left(\frac23\ln N+1\right)S_T\Delta_T.}\]
\end{frame}

\begin{frame}{Шаг 4. Решение квадратного неравенства}
  Обозначим
  \[a=V_T\ln N,\qquad b=S_T\left(\frac23\ln N+1\right).\]
  Из $\Delta_T^2\le a+b\Delta_T$ и $a,b\ge0$ следует
  \[
    \Delta_T\le\frac{b+\sqrt{b^2+4a}}2\le\sqrt a+b.
  \]
  Подставляя в промежуточную гарантию, получаем
  \[
    \widetilde R_T^i\le R_T^i\le2\Delta_T
      \le2\sqrt{V_T\ln N}+S_T\left(\frac43\ln N+2\right).
  \]
  \par\medskip\textbf{Граница следует из контроля накопленного разрыва между $\ell_t$ и $m_t$.}
\end{frame}

\begin{frame}{Переход от смеси потерь к прогнозам}
  Пусть $\mathcal D$ выпукло, а $\ell(\gamma,y)$ выпукла по $\gamma$.
  Агрегатор выдаёт
  \[\gamma_t=\sum_{i\in E_t} p_t^{i,E_t}\gamma_t^i.\]
  По Йенсену фактическая потеря не больше смеси потерь:
  \[
    \ell(\gamma_t,y_t)\le\sum_{i\in E_t} p_t^{i,E_t}\ell(\gamma_t^i,y_t)=\ell_t.
  \]
  \begin{block}{Определение 19. Регрет прогноза на активных шагах}
    \[
      \widetilde R_{T,\mathrm{pred}}^i
      :=\sum_{\substack{1\le t\le T\\i\in E_t}}
          \bigl(\ell(\gamma_t,y_t)-\ell(\gamma_t^i,y_t)\bigr).
    \]
  \end{block}
  \textbf{Следствие:} $\widetilde R_{T,\mathrm{pred}}^i\le\widetilde R_T^i$;
  оценка теоремы сохраняется для прогнозов.
  \par\vfill{\scriptsize Суммарные фактические потери прогноза: $A_T=\sum_t\ell(\gamma_t,y_t)$.\par}
\end{frame}

\section{Резерв}
\begin{frame}[plain,noframenumbering]{Резерв A1. Монотонность потенциала по параметру}
  Зафиксируем потери $L_T^i$ и начальные веса $w_1^i$, $\sum_iw_1^i=1$.
  Пусть $0<\eta\le\mu$ и $r=\eta/\mu\in(0,1]$.
  \[
    Z_\eta:=\sum_i w_1^i e^{-\eta L_T^i}
       =\sum_i w_1^i\left(e^{-\mu L_T^i}\right)^r.
  \]
  По вогнутости $x\mapsto x^r$:
  \[Z_\eta\le\left(\sum_i w_1^i e^{-\mu L_T^i}\right)^r=Z_\mu^r.\]
  Следовательно,
  \[
    M_T^\eta=-\frac1\eta\ln Z_\eta
       \ge-\frac r\eta\ln Z_\mu=M_T^\mu.
  \]
  \par\medskip\textbf{Кумулятивная смешанная потеря невозрастает по $\eta$.}
  \par\vfill{\scriptsize При сравнении параметров последовательность потерь фиксирована, включая уже полученные виртуальные потери.\par}
\end{frame}

\begin{frame}[plain,noframenumbering]{Резерв A2. Оценка при невозрастающем параметре}
  Из определения весов при текущем $\eta_t$ имеем
  \[m_t=M_t^{\eta_t}-M_{t-1}^{\eta_t},\qquad M_0^\eta=0.\]
  Суммируем и группируем члены:
  \begin{align*}
    M_T
      &=\sum_{t=1}^T\left(M_t^{\eta_t}-M_{t-1}^{\eta_t}\right)\\
      &=M_T^{\eta_T}
         +\sum_{t=1}^{T-1}\left(M_t^{\eta_t}-M_t^{\eta_{t+1}}\right).
  \end{align*}
  При $\eta_t\ge\eta_{t+1}$ монотонность даёт
  $M_t^{\eta_t}\le M_t^{\eta_{t+1}}$.
  \[\alert{M_T\le M_T^{\eta_T}\le L_T^i+\frac{\ln N}{\eta_T}.}\]
\end{frame}

\begin{frame}[plain,noframenumbering]{Резерв A3. Начальные и вырожденные случаи}
  \textbf{Почему при $\Delta_{t-1}=0$ расширенные потери равны?}
  \begin{itemize}
    \item Изначально все $L_0^i=0$.
    \item При равных $L_{t-1}^i$ активные веса равномерны.
    \item Если $\delta_t=0$, среднее активных потерь равно их минимуму.
          Значит, все эти потери равны $m_t$.
    \item Спящим экспертам тоже присваивается $m_t$:
          все $L_t^i$ остаются равными.
  \end{itemize}
  \medskip
  \textbf{Совпадение активных потерь.}
  При $s_t=0$, в том числе при $|E_t|=1$, получаем
  $m_t=\ell_t$, $\delta_t=v_t=0$.
  \medskip
  \textbf{Пустое $E_t$.} Нормировка не определена.
  Протокол требует хотя бы одного доступного эксперта на каждом шаге.
\end{frame}

\begin{frame}[plain,noframenumbering]{Резерв A4. Применение неравенства Бернштейна}
  Для $X\in[0,1]$ и $a>0$ используем
  \[\ln\mathbb{E} e^{-a(X-\mathbb{E} X)}\le\operatorname{Var}(X)(e^a-1-a).\]
  При $s_t>0$ положим $\ell_t^-=\min_{i\in E_t}\ell_t^i$ и зададим
  \[
    X=\frac{\ell_t^i-\ell_t^-}{s_t}
       \quad\text{с вероятностью } p_t^{i,E_t},\qquad a=s_t\eta_t.
  \]
  Тогда $\mathbb{E} X=(\ell_t-\ell_t^-)/s_t$, $\operatorname{Var}(X)=v_t/s_t^2$ и
  \begin{align*}
    \eta_t\delta_t
       &=\ln\sum_{i\in E_t} p_t^{i,E_t} e^{-\eta_t(\ell_t^i-\ell_t)}\\
       &=\ln\mathbb{E} e^{-s_t\eta_t(X-\mathbb{E} X)}
         \le\frac{v_t}{s_t^2}\left(e^{s_t\eta_t}-1-s_t\eta_t\right).
  \end{align*}
  Делим на $\eta_t>0$ и получаем одношаговую оценку.
  \par\vfill{\scriptsize Неравенство Бернштейна в этой форме: Cesa-Bianchi--Lugosi, леммы A.4--A.5~\cite{CBL}.\par}
\end{frame}

\begin{frame}[plain,noframenumbering]{Резерв A5. Переход к оценке отношения смешанной разности}
  Обозначим $x=s_t\eta_t>0$. Если $0<x<3$, то
  \[
    e^x-1-x=\sum_{k=2}^{\infty}\frac{x^k}{k!}
      \le\frac{x^2}{2}\sum_{j=0}^{\infty}(x/3)^j
      =\frac{x^2}{2(1-x/3)}.
  \]
  Здесь $k!\ge2\cdot3^{k-2}$ при $k\ge2$. Поэтому
  \[
    \delta_t\le\frac{\eta_t v_t}{2(1-s_t\eta_t/3)}
    \quad\Longrightarrow\quad
    \frac{\delta_t}{\eta_t}\le\frac12v_t+\frac13s_t\delta_t.
  \]
  Если $x\ge3$, то непосредственно
  \[
    \frac{\delta_t}{\eta_t}=\frac{s_t\delta_t}{x}
       \le\frac13s_t\delta_t
       \le\frac12v_t+\frac13s_t\delta_t.
  \]
  \par\vfill{\scriptsize Таким образом, неравенство верно для всех конечных $\eta_t>0$, а не только при малом произведении $s_t\eta_t$.\par}
\end{frame}
